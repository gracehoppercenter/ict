DIGITS = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"


def encode(s):
    """
      >>> encode("W1sT/P")
      [22, 53, 44, 19, 63, 15]
      >>> encode("ABCDEFG")
      [0, 1, 2, 3, 4, 5, 6]
       encode("abcdefg")
      [26, 27, 28, 29, 30, 31, 32]
      >>> encode("12345")
      [53, 54, 55, 56, 57]
    """
    ints = []
    for ch in s:
        ints.append(DIGITS.index(ch))
    return ints


def decode(nums):
    """
      >>> decode([22, 53, 44, 19, 63, 15])
      'W1sT/P'
      >>> decode([0, 1, 2, 3, 4, 5, 6])
      'ABCDEFG'
      >>> decode([26, 27, 28, 29, 30, 31, 32])
      'abcdefg'
      >>> decode([53, 54, 55, 56, 57])
      '12345'
    """
    s = ""
    for i in nums:
        s += DIGITS[i]
    return s


if __name__ == '__main__':
    import doctest
    doctest.testmod()
